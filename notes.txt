Task-2: Firewall.py:
import requests
import csv
import subprocess

# Take rule name from user
rule_name = input("Enter firewall rule name: ")

# Download IP list
response = requests.get(
    "https://feodotracker.abuse.ch/downloads/ipblocklist.csv"
).text

# Delete old rule
delete_rule = f'netsh advfirewall firewall delete rule name="{rule_name}"'
subprocess.run(["PowerShell", "-Command", delete_rule])

# Read CSV
mycsv = csv.reader(
    filter(lambda x: not x.startswith("#"), response.splitlines())
)

# Add blocking rules
for row in mycsv:
    ip = row[1]

    if ip != "dst_ip":
        print("Blocking IP:", ip)

        add_rule = (
            f'netsh advfirewall firewall add rule '
            f'name="{rule_name}" Dir=Out Action=Block RemoteIP={ip}'
        )

        subprocess.run(["PowerShell", "-Command", add_rule])

print("Done!")

Task:3- Passwordchecker.py
import re

def check_password(password):
    if len(password) < 8:
        return "Need at least 8 characters"
    elif not re.search(r"\d", password):
        return "Need a number"
    elif not re.search(r"[A-Z]", password):
        return "Need an uppercase letter"
    elif not re.search(r"[a-z]", password):
        return "Need a lowercase letter"
    elif not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        return "Need a special character"
    else:
        return "Strong password!"

while True:
    pwd = input("Enter password (or 'exit'): ")
    
    if pwd.lower() == "exit":
        print("Goodbye!")
        break
    
    print(check_password(pwd))
